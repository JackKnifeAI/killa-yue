cmake_minimum_required(VERSION 3.22)
project(hammer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(OPUS_DIR)
  set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
  set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${OPUS_DIR} opus EXCLUDE_FROM_ALL)
  set(OPUS_TARGET opus)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
  set(OPUS_TARGET PkgConfig::OPUS)
endif()
if(NOT ANDROID)
  find_package(JNI REQUIRED)
endif()
add_library(hammer SHARED codec.cpp ../../studio/native/src/resample.cpp)
target_include_directories(hammer SYSTEM PRIVATE vendor ${JNI_INCLUDE_DIRS})
target_include_directories(hammer PRIVATE ../../studio/native/include)
if(OPUS_DIR)
  target_include_directories(hammer SYSTEM PRIVATE ${OPUS_DIR}/include)
endif()
target_link_libraries(hammer PRIVATE ${OPUS_TARGET})
if(MSVC)
  target_compile_options(hammer PRIVATE /O2 /W3)
  target_compile_definitions(hammer PRIVATE NOMINMAX)
else()
  target_compile_options(hammer PRIVATE -O2 -Wall -Wextra)
endif()
if(ANDROID)
  target_link_libraries(hammer PRIVATE log)
  target_link_options(hammer PRIVATE -Wl,-z,max-page-size=16384)
else()
  target_link_libraries(hammer PRIVATE ${CMAKE_DL_LIBS})
endif()
